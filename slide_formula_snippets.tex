\section*{Influence Score}

\begin{equation*}
\begin{aligned}
\mathrm{Influence\_Score}(w)
&= 
1.0 \cdot \mathbf{1}[\text{answer changed}]
\\
&\quad + 0.5 \cdot \mathbf{1}[\text{abstention flipped}]
\\
&\quad + 0.25 \cdot \mathbf{1}[\text{retrieved set changed}]
\\
&\quad + 0.25 \cdot \mathbf{1}[\text{prompt set changed}]
\\
&\quad + 0.25 \cdot \mathrm{answer\ distance}
\end{aligned}
\end{equation*}

\section*{Gini Coefficient}

\[
Gini\_Coefficient(x_1,\dots,x_n)
=
\frac{2 \sum_{i=1}^{n} i x_{(i)}}{n \sum_{i=1}^{n} x_i}
- \frac{n+1}{n}
\]
